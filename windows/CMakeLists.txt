function(spc_shader entry profile)
    set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/${entry}.h")
    add_custom_command(OUTPUT "${output}"
        COMMAND "${SPC_FXC}" /nologo /T ${profile} /E ${entry}
            /Fh "${output}" /Vn ${entry}Bytecode "${CMAKE_CURRENT_SOURCE_DIR}/src/render/shaders/Memory.hlsl"
        DEPENDS src/render/shaders/Memory.hlsl VERBATIM)
    set(SPC_SHADER_HEADERS ${SPC_SHADER_HEADERS} "${output}" PARENT_SCOPE)
endfunction()
spc_shader(FullscreenVS vs_5_0)
spc_shader(MapPS ps_5_0)
spc_shader(AreaPS ps_5_0)
spc_shader(WaterfallPS ps_5_0)
spc_shader(WaterfallAreaPS ps_5_0)
spc_shader(BlitPS ps_5_0)
foreach(entry SpectrumProjectCS SpectrumDisplayPS)
    if(entry STREQUAL "SpectrumProjectCS")
        set(profile cs_5_0)
    else()
        set(profile ps_5_0)
    endif()
    set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/${entry}.h")
    add_custom_command(OUTPUT "${output}"
        COMMAND "${SPC_FXC}" /nologo /T ${profile} /E ${entry} /Fh "${output}" /Vn ${entry}Bytecode
        "${CMAKE_CURRENT_SOURCE_DIR}/src/render/shaders/Spectrogram.hlsl"
        DEPENDS src/render/shaders/Spectrogram.hlsl VERBATIM)
    list(APPEND SPC_SHADER_HEADERS "${output}")
endforeach()

if(SPC_ENABLE_IPO)
    check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error LANGUAGES C CXX)
    if(NOT ipo_supported)
        message(FATAL_ERROR "IPO unavailable: ${ipo_error}; configure SPC_ENABLE_IPO=OFF to diagnose")
    endif()
    foreach(target spc_backend spc_web_core spc_native_core spc_web_render spc_desktop_models spc_windows_audio spc_graphics VisualSPCPlayer spc_spectrum_benchmark)
        set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    endforeach()
endif()
